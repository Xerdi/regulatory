\loadsources{sources}
% De derde route: een bron die in het document zelf staat, zonder bestand.
\newsource{uavg}{regeling}{%
    citeertitel = Uitvoeringswet Algemene verordening gegevensbescherming,
    afkorting = UAVG,
    lidwoord = de%
}
% Dezelfde verordening zoals een Duits en een Frans document haar aanhalen: het
% register hoort bij de bron, dus de bron draagt haar naam in de taal waarin ze
% aangehaald wordt. Deze twee gebruiken de Engelse veldnamen, de rest de
% Nederlandse; beide horen te werken.
\newsource{avgde}{euact}{%
    kind = Verordnung, number = {(EU) 2016/679},
    shortname = Datenschutz-Grundverordnung, abbreviation = DSGVO,
    register = german_eu%
}
\newsource{avgfr}{euact}{%
    kind = Règlement, number = {(UE) 2016/679},
    shortname = {règlement général sur la protection des données},
    abbreviation = RGPD, register = french_eu%
}
% En zoals een Engels document haar aanhaalt, wat het enige register is dat de
% delen andersom zet: het diepste niveau eerst.
\newsource{avgen}{euact}{%
    kind = Regulation, number = {(EU) 2016/679},
    shortname = General Data Protection Regulation, abbreviation = GDPR,
    determiner = the, register = english_eu%
}
% Een verdrag krijgt in het Engels wel een lidwoord en een verordening niet, en
% dat volgt uit de soort en niet uit het register.
\newsource{vweu}{regulation}{%
    citetitle = Treaty on the Functioning of the European Union,
    kind = Treaty, abbreviation = TFEU, register = english_eu%
}
\newsource{wetib}{regeling}{%
    citeertitel = Wet inkomstenbelasting 2001,
    afkorting = Wet IB 2001,
    lidwoord = de%
}

\hypersetup{pdftitle={Voorbeeld Bronnen}}

% Wat de suite leest. Een veld dat verkeerd is ingelezen valt in de PDF niet op,
% dus wordt het hier genoemd waar een controle erbij kan.
\newcommand\report[2]{\typeout{SRC #1 #2 = \srcfield{#1}{#2}}}
% Hetzelfde voor een verwijzing: die wordt eerst opgebouwd en dan gezet, dus het
% opgebouwde is wat een controle kan lezen. Een verwijzing in het verkeerde
% register faalt nergens, hij staat er gewoon verkeerd.
\makeatletter
\newcommand\citereport[2]{#1\typeout{CITE #2 = \regulatory@src@last}}
\makeatother
